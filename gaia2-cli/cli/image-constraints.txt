# rebuild of the same commit resolves the newest litellm and dependencies, and

# completed end-to-end judged runs. To refresh them, resolve without pins

#   pip install --dry-run --ignore-installed --report report.json \
#       ./core './cli[judge]' 'huggingface_hub[cli]'
